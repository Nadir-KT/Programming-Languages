/*
Create structure and pass it in function and use the elements inside function. ( use pointer )
*/
#include <stdio.h>

typedef struct {
	int ele1;
	int ele2;
} stu;

void fun(stu *a);

void main(){
stu stu1;

stu *stu2=&stu1;
 
stu2->ele1=12;
stu2->ele2=13;

fun(stu2);
}

void fun(stu *a)
{
	printf("%d",a->ele1);
	printf("%d",a->ele2);
}
