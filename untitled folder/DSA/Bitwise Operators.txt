write a c program to find counts of 0's and 1's in 0x86

#include <stdio.h>

// Function to count 0's and 1's in binary representation
void countZeroesOnes(int num) {
    int count_0 = 0, count_1 = 0;
    
    // Iterate through each bit position
    for (int i = 0; i < 8; ++i) { // 0x86 is an 8-bit number (1 byte)
        if ((num & (1 << i)) == 0)
            count_0++;
        else
            count_1++;
    }
    
    // Print results
    printf("Number of 0's in binary representation: %d\n", count_0);
    printf("Number of 1's in binary representation: %d\n", count_1);
}

int main() {
    // Hexadecimal number 0x86
    int hex_number = 0x86;
    
    // Count 0's and 1's in binary representation
    countZeroesOnes(hex_number);
    
    return 0;
}
